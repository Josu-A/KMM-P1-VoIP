% Theorems
\usepackage[framemethod=default]{mdframed}
\newcounter{theorem}[section]
\newenvironment{theorem}[1][]{%
    \stepcounter{theorem}%
    \ifstrempty{#1}%
    {\mdfsetup{%
        frametitle={%
            \tikz[baseline=(current bounding box.east),outer sep=0pt]
            \node[anchor=east,rectangle,fill=black!20]{\strut Teorema \thetheorem};
        }
    }}%
    {\mdfsetup{%
        frametitle={%
            \tikz[baseline=(current bounding box.east),outer sep=0pt]
            \node[anchor=east,rectangle,fill=black!20]{\strut Teorema \thetheorem: #1};
        }
    }}%
    \mdfsetup{
        innertopmargin=10pt,%
        linecolor=black!20,%
        linewidth=2pt,%
        topline=true,%
        frametitleaboveskip=\the\dimexpr -\ht\strutbox \relax,%
        skipabove=\the\baselineskip,
    }
    \begin{mdframed}[]\relax%
}{
    \end{mdframed}
}

% Important
\newenvironment{important}[1][]{%
   \begin{mdframed}[%
      backgroundcolor={red!15}, hidealllines=true,
      skipabove=0.7\baselineskip, skipbelow=0.7\baselineskip,
      splitbottomskip=2pt, splittopskip=4pt, #1]%
   \makebox[0pt]{% ignore the withd of !
      \smash{% ignor the height of !
         \fontsize{32pt}{32pt}\selectfont% make the ! bigger
         \hspace*{-19pt}% move ! to the left
         \raisebox{-2pt}{% move ! up a little
            {\color{red!70!black}\sffamily\bfseries !}% type the bold red !
         }%
      }%
   }%
}{\end{mdframed}}
